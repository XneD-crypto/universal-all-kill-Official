-- Функция обновления кнопки Save
local function updateSaveButton()
    if not SaveNickBtn then return end
    local savedList = {}
    for name, _ in pairs(savedPlayers) do
        table.insert(savedList, name)
    end
    if #savedList > 0 then
        local display = savedList[1] or ""
        if #savedList > 1 then
            display = display .. ", " .. (savedList[2] or "")
        end
        if #savedList > 2 then
            display = display .. " и тд"
        end
        SaveNickBtn.Text = "Saved: " .. display
        SaveNickBtn.BackgroundColor3 = Color3.fromRGB(0, 100, 200)
    else
        SaveNickBtn.Text = "Save \"Nicname\""
        SaveNickBtn.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
    end
end

-- Создание купола для игрока
local function createDomeForPlayer(player)
    if activeDomes[player.Name] then return end
    if not player.Character then return end
    local hrp = player.Character:FindFirstChild("HumanoidRootPart")
    if not hrp then return end

    local visual = Instance.new("Part")
    visual.Shape = Enum.PartType.Ball
    visual.Size = Vector3.new(10, 10, 10)
    visual.Transparency = 0.7
    visual.Color = Color3.fromRGB(0, 150, 255)
    visual.Material = Enum.Material.ForceField
    visual.CanCollide = false
    visual.Anchored = true
    visual.Parent = workspace

    local physical = Instance.new("Part")
    physical.Shape = Enum.PartType.Ball
    physical.Size = Vector3.new(10, 10, 10)
    physical.Transparency = 1
    physical.CanCollide = true
    physical.Anchored = true
    physical.Parent = workspace

    activeDomes[player.Name] = {visual = visual, physical = physical, player = player}
end

local function removeDome(name)
    local dome = activeDomes[name]
    if dome then
        if dome.visual then dome.visual:Destroy() end
        if dome.physical then dome.physical:Destroy() end
        activeDomes[name] = nil
    end
end

local function isPlayerOnServer(name)
    for _, p in pairs(plrs:GetPlayers()) do
        if p.Name == name then return true end
    end
    return false
end

local function cleanupMissingPlayers()
    for name, _ in pairs(savedPlayers) do
        if not isPlayerOnServer(name) then
            savedPlayers[name] = nil
            removeDome(name)
        end
    end
    for name, _ in pairs(selectedForKill) do
        if not isPlayerOnServer(name) then
            selectedForKill[name] = nil
        end
    end
end

-- ===== ОСТАЛЬНАЯ ЛОГИКА =====
local function isAnyPlayerPart(p)
    for _, plr in pairs(plrs:GetPlayers()) do
        if plr.Character then
            if p:IsDescendantOf(plr.Character) or p.Parent == plr.Character then
                return true
            end
        end
    end
    return false
end

local function isNearLocalPlayer(p)
    if not lplr.Character then return false end
    local root = lplr.Character:FindFirstChild("HumanoidRootPart")
    if not root then return false end
    return (p.Position - root.Position).Magnitude < 20
end

local function isValidPart(p)
    if not p:IsA("BasePart") then return false end
    if p.Anchored then return false end
    if not p:IsDescendantOf(workspace) then return false end
    if isAnyPlayerPart(p) then return false end
    return true
end

local function collectParts()
    parts = {}
    for _, p in pairs(workspace:GetDescendants()) do
        if isValidPart(p) and not table.find(parts, p) then
            table.insert(parts, p)
        end
    end
end
collectParts()

workspace.DescendantAdded:Connect(function(p)
    task.wait(0.05)
    if isValidPart(p) and not table.find(parts, p) then
        table.insert(parts, p)
    end
end)

workspace.DescendantRemoving:Connect(function(p)
    local i = table.find(parts, p)
    if i then table.remove(parts, i) end
end)

local function getTargetPlayer(name)
    if not name or name == "" then return nil end
    local lowerTarget = name:lower()
    for _, p in pairs(plrs:GetPlayers()) do
        if p ~= lplr then
            if p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
                if not table.find(blacklist, p.Name) and not table.find(saveOnlyBlacklist, p.Name) then
                    if p.Name:lower():sub(1, #lowerTarget) == lowerTarget or p.DisplayName:lower():sub(1, #lowerTarget) == lowerTarget then
                        return p
                    end
                end
            end
        end
    end
    return nil
end

local function getAllPlayers()
    local allPlayers = {}
    if next(selectedForKill) ~= nil then
        for name, _ in pairs(selectedForKill) do
            local p = getTargetPlayer(name)
            if p then table.insert(allPlayers, p) end
        end
        return allPlayers
    end

    if targetName ~= "" then
        local tp = getTargetPlayer(targetName)
        if tp then return {tp} end
        return {}
    end
    
    for _, p in pairs(plrs:GetPlayers()) do
        if p ~= lplr and p.Character and not table.find(blacklist, p.Name) and not table.find(saveOnlyBlacklist, p.Name) and not savedPlayers[p.Name] then
            local troot = p.Character:FindFirstChild("HumanoidRootPart")
            if troot then
                table.insert(allPlayers, p)
            end
        end
    end
    return allPlayers
end

local function start()
    if connection then return end
    connection = rs.Heartbeat:Connect(function()
        -- Обновляем позиции куполов
        for name, dome in pairs(activeDomes) do
            local player = dome.player
            if player and player.Character then
                local hrp = player.Character:FindFirstChild("HumanoidRootPart")
                if hrp then
                    local pos = hrp.Position
                    if dome.visual then dome.visual.CFrame = CFrame.new(pos) end
                    if dome.physical then dome.physical.CFrame = CFrame.new(pos) end
                end
            end
        end

        -- Основная логика полёта блоков
        for _, p in pairs(parts) do
            if p and p.Parent and not p.Anchored and isNearLocalPlayer(p) then
                p.CanCollide = false
            end
        end
        
        local currentTime = tick()
        local allPlayers = getAllPlayers()

        if #allPlayers > 0 then
            if currentTime - lastRotation >= rotationspeed then
                currentTargetIndex = currentTargetIndex + 1
                if currentTargetIndex > #allPlayers then
                    currentTargetIndex = 1
                end
                lastRotation = currentTime
            end

            if split then
                local numTargets = math.min(maxtargets, #allPlayers)
                local targets = {}

                for i = 1, numTargets do
                    local idx = ((currentTargetIndex + i - 2) % #allPlayers) + 1
                    local t = allPlayers[idx]
                    if t and t.Character then
                        local root = t.Character:FindFirstChild("HumanoidRootPart")
                        if root then
                            table.insert(targets, root)
                        end
                    end
                end

                for i, p in pairs(parts) do
                    if p and p.Parent and not p.Anchored and #targets > 0 then
                        if isNearLocalPlayer(p) then continue end
                        local targetIdx = ((i - 1) % #targets) + 1
                        local center = targets[targetIdx].Position
                        local pos = p.Position
                        local ang = math.atan2(pos.Z - center.Z, pos.X - center.X)
                        local nang = ang + math.rad(10)
                        local tpos = Vector3.new(
                            center.X + math.cos(nang) * 10,
                            center.Y + (0.1 * (math.abs(math.sin((pos.Y - center.Y) / 0.1)))),
                            center.Z + math.sin(nang) * 10
                        )
                        p.Velocity = (tpos - p.Position).Unit * 1000
                    end
                end
            else
                local t = allPlayers[currentTargetIndex]
                if t and t.Character then
                    local root = t.Character:FindFirstChild("HumanoidRootPart")
                    if root then
                        local center = root.Position
                        for _, p in pairs(parts) do
                            if p and p.Parent and not p.Anchored then
                                if isNearLocalPlayer(p) then continue end
                                local pos = p.Position
                                local ang = math.atan2(pos.Z - center.Z, pos.X - center.X)
                                local nang = ang + math.rad(10)
                                local tpos = Vector3.new(
                                    center.X + math.cos(nang) * 10,
                                    center.Y + (0.1 * (math.abs(math.sin((pos.Y - center.Y) / 0.1)))),
                                    center.Z + math.sin(nang) * 10
                                )
                                p.Velocity = (tpos - p.Position).Unit * 1000
                            end
                        end
                    end
                end
            end
        end

        -- Проверка куполов (только для тех блоков, которые внутри или касаются)
        for name, dome in pairs(activeDomes) do
            if dome.physical and dome.physical.Parent then
                local center = dome.physical.Position
                local radius = 5.5
                for _, p in pairs(parts) do
                    if p and p.Parent and not p.Anchored then
                        local dist = (p.Position - center).Magnitude
                        if dist <= radius then
                            local direction = (p.Position - center).Unit
                            if direction.Magnitude == 0 then direction = Vector3.new(0, 1, 0) end
                            p.Velocity = direction * 5000 + Vector3.new(0, 5000, 0)
                        end
                    end
                end
            end
        end
    end)
    print("[FE Kill] Enabled")
end

local function stop()
    if connection then
        connection:Disconnect()
        connection = nil
        currentTargetIndex = 1
        lastRotation = 0
    end
    for _, p in pairs(parts) do
        if p and p.Parent and not p.Anchored and p:IsDescendantOf(workspace) then
            p.Velocity = Vector3.zero
            p.RotVelocity = Vector3.zero
        end
    end
    for name, dome in pairs(activeDomes) do
        if dome.visual then dome.visual:Destroy() end
        if dome.physical then dome.physical:Destroy() end
    end
    activeDomes = {}
    print("[FE Kill] Disabled")
end

lplr.Chatted:Connect(function(msg)
    local cmd = msg:lower()
    
    if cmd == "/died off" then
        if enabled then
            enabled = false
            stop()
        end
        return
    end
    
    if cmd == "/died all" then
        targetName = ""
        if not enabled then
            enabled = true
            start()
        else
            stop()
            start()
        end
        return
    end
    
    local nickNoQuote = cmd:match('^/died%s+(.+)$')
    if nickNoQuote and nickNoQuote ~= "all" and nickNoQuote ~= "off" then
        targetName = nickNoQuote
        if not enabled then
            enabled = true
            start()
        else
            stop()
            start()
        end
        return
    end
end)

local function makeDraggable(frame)
    local dragging, dragStart, startPos
    frame.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = frame.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)
    frame.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch and dragging then
            local delta = input.Position - dragStart
            frame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end

-- ===== СИСТЕМА МУЛЬТИВЫБОРА =====
local function openMultiSelector(actionType)
    local SelectorGui = Instance.new("ScreenGui")
    SelectorGui.Name = "PlayerSelector"
    SelectorGui.Parent = lplr.PlayerGui
    SelectorGui.ResetOnSpawn = false
    SelectorGui.DisplayOrder = 1000

    local Background = Instance.new("Frame")
    Background.Size = UDim2.new(0, 250, 0, 300)
    Background.Position = UDim2.new(0.5, -125, 0.5, -150)
    Background.BackgroundColor3 = Color3.fromRGB(20, 20, 20)
    Background.Parent = SelectorGui

    local BgCorner = Instance.new("UICorner")
    BgCorner.CornerRadius = UDim.new(0, 10)
    BgCorner.Parent = Background

    local Title = Instance.new("TextLabel")
    Title.Size = UDim2.new(1, 0, 0, 30)
    Title.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
    Title.Text = actionType == "save" and "Сохранение (нажми для вкл/выкл)" or "Выбор цели (нажми для вкл/выкл)"
    Title.TextColor3 = Color3.fromRGB(255, 255, 255)
    Title.Font = Enum.Font.GothamBold
    Title.TextSize = 14
    Title.Parent = Background

    local TitleCorner = Instance.new("UICorner")
    TitleCorner.CornerRadius = UDim.new(0, 10)
    TitleCorner.Parent = Title

    local BackBtn = Instance.new("TextButton")
    BackBtn.Size = UDim2.new(0, 24, 0, 24)
    BackBtn.Position = UDim2.new(0, 5, 0, 3)
    BackBtn.BackgroundColor3 = Color3.fromRGB(60, 60, 60)
    BackBtn.Text = "←"
    BackBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
    BackBtn.TextSize = 16
    BackBtn.Font = Enum.Font.GothamBold
    BackBtn.BorderSizePixel = 0
    BackBtn.Parent = Title

    local BackCorner = Instance.new("UICorner")
    BackCorner.CornerRadius = UDim.new(1, 0)
    BackCorner.Parent = BackBtn

    BackBtn.MouseButton1Click:Connect(function()
        SelectorGui:Destroy()
        Main.Visible = true
        Main.Size = UDim2.new(0, 250, 0, 200)
        Main.Position = UDim2.new(0.5, -125, 0.5, -100)
        isMenuOpen = true
        ToggleBtn.Text = "🎉"
    end)

    local Scroll = Instance.new("ScrollingFrame")
    Scroll.Size = UDim2.new(1, -10, 1, -40)
    Scroll.Position = UDim2.new(0, 5, 0, 35)
    Scroll.BackgroundTransparency = 1
    Scroll.ScrollBarThickness = 4
    Scroll.ScrollBarImageColor3 = Color3.fromRGB(100, 100, 100)
    Scroll.AutomaticCanvasSize = Enum.AutomaticSize.Y
    Scroll.Parent = Background

    local List = Instance.new("UIListLayout")
    List.Padding = UDim.new(0, 5)
    List.Parent = Scroll

    for _, p in pairs(plrs:GetPlayers()) do
        if p ~= lplr then
            local show = true
            if table.find(blacklist, p.Name) then
                show = false
            elseif actionType == "kill" and table.find(saveOnlyBlacklist, p.Name) then
                show = false
            end

            if show then
                local Btn = Instance.new("TextButton")
                Btn.Size = UDim2.new(1, 0, 0, 40)
                local isSelected = (actionType == "save" and savedPlayers[p.Name]) or (actionType == "kill" and selectedForKill[p.Name])
                Btn.BackgroundColor3 = isSelected and Color3.fromRGB(0, 100, 200) or Color3.fromRGB(35, 35, 35)
                Btn.Text = ""
                Btn.Parent = Scroll

                local BtnCorner = Instance.new("UICorner")
                BtnCorner.CornerRadius = UDim.new(0, 6)
                BtnCorner.Parent = Btn

                local Avatar = Instance.new("ImageLabel")
                Avatar.Size = UDim2.new(0, 30, 0, 30)
                Avatar.Position = UDim2.new(0, 5, 0, 5)
                Avatar.BackgroundColor3 = Color3.fromRGB(50, 50, 50)
                Avatar.Image = "https://www.roblox.com/headshot-thumbnail/image?userId=" .. p.UserId .. "&width=150&height=150&format=png"
                Avatar.Parent = Btn

                local AvatarCorner = Instance.new("UICorner")
                AvatarCorner.CornerRadius = UDim.new(1, 0)
                AvatarCorner.Parent = Avatar

                local NameLbl = Instance.new("TextLabel")
                NameLbl.Size = UDim2.new(1, -45, 1, 0)
                NameLbl.Position = UDim2.new(0, 40, 0, 0)
                NameLbl.BackgroundTransparency = 1
                NameLbl.Text = p.Name
                NameLbl.TextColor3 = Color3.fromRGB(255, 255, 255)
                NameLbl.Font = Enum.Font.Gotham
                NameLbl.TextSize = 12
                NameLbl.TextXAlignment = Enum.TextXAlignment.Left
                NameLbl.Parent = Btn

                Btn.MouseButton1Click:Connect(function()
                    if actionType == "save" then
                        if savedPlayers[p.Name] then
                            savedPlayers[p.Name] = nil
                            Btn.BackgroundColor3 = Color3.fromRGB(35, 35, 35)
                            removeDome(p.Name)
                        else
                            savedPlayers[p.Name] = true
                            Btn.BackgroundColor3 = Color3.fromRGB(0, 100, 200)
                            createDomeForPlayer(p)
                        end
                        updateSaveButton()
                    else
                        if selectedForKill[p.Name] then
                            selectedForKill[p.Name] = nil
                            Btn.BackgroundColor3 = Color3.fromRGB(35, 35, 35)
                        else
                            selectedForKill[p.Name] = true
                            Btn.BackgroundColor3 = Color3.fromRGB(0, 100, 200)
                        end
                        local killList = {}
                        for name, _ in pairs(selectedForKill) do table.insert(killList, name) end
                        if #killList > 0 then
                            local display = killList[1] or ""
                            if #killList > 1 then display = display .. ", " .. (killList[2] or "") end
                            if #killList > 2 then display = display .. " и тд" end
                            DiedNickBtn.Text = "Died: " .. display
                            DiedNickBtn.BackgroundColor3 = Color3.fromRGB(0, 150, 0)
                            if not enabled then
                                enabled = true
                                start()
                            end
                        else
                            DiedNickBtn.Text = "Died \"nicname\""
                            DiedNickBtn.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
                        end
                    end
                end)
            end
        end
    end
end

DiedNickBtn.MouseButton1Click:Connect(function()
    Main.Visible = false
    openMultiSelector("kill")
end)

SaveNickBtn.MouseButton1Click:Connect(function()
    Main.Visible = false
    openMultiSelector("save")
end)
